### 164. Maximum Gap


Given an integer array nums, return the maximum difference between two successive elements in its sorted form. If the array contains less than two elements, return 0.

You must write an algorithm that runs in linear time and uses linear extra space.

 

Example 1:

Input: nums = [3,6,9,1]
Output: 3
Explanation: The sorted form of the array is [1,3,6,9], either (3,6) or (6,9) has the maximum difference 3.
Example 2:

Input: nums = [10]
Output: 0
Explanation: The array contains less than 2 elements, therefore return 0.


#### Solution 1

```cpp
class Solution {
public:
    int maximumGap(vector<int>& nums) {
        if (nums.size() < 2) 
            return 0;
        
        auto min_max_pair = std::minmax_element(nums.begin(), nums.end());
        int lo = *min_max_pair.first;
        int hi = *min_max_pair.second;
        int bucket_size = max(1, int((hi - lo) / (nums.size() - 1)));
        vector<vector<int>> buckets((hi - lo) / bucket_size + 1, vector<int>());
        for (auto& num : nums){
            buckets[(num - lo) / bucket_size].push_back(num);
        }
        int curr_hi = 0;
        int result = 0;
        for (auto& bucket : buckets){
            if (bucket.empty()) 
                continue;
            int prev_hi = curr_hi ? curr_hi : bucket[0];
            int curr_lo = bucket[0];
            for (auto& num : bucket){
                curr_hi = max(curr_hi, num);
                curr_lo = min(curr_lo, num);
            }
            result = max(result, curr_lo - prev_hi);
        }
        return result;
    }
};
```

#### Optimized solution 2

```cpp
class Solution {
public:
    int maximumGap(vector<int>& nums) {
        int n=nums.size();
        if(n<2)
            return 0;
        int mn=INT_MAX,mx=INT_MIN;
        for(int num:nums){
            mn=min(num,mn);
            mx=max(num,mx);
        }
        if(mx==mn)
            return 0;
        if(n==2)
            return mx-mn;
        int bs=ceil((double)(mx-mn)/(n-1));
        int bc=(mx-mn)/(bs);
        vector<pair<int,int>> b(bc+1,pair<int,int>(INT_MAX,INT_MIN));
        for(int num:nums){
            int i=(num-mn)/bs;
            b[i].first=min(num,b[i].first);
            b[i].second=max(num,b[i].second);
        }
        int diff=0,m=b[0].second;
        for(int i=1;i<=bc;i++){
            if(b[i].first!=INT_MAX && m!=INT_MIN){
                diff=max(diff,b[i].first-m);
            }
            if(b[i].second!=INT_MIN)
                m=b[i].second;
        }
        return diff;
    }
};
```
