# 126. Word Ladder II
A transformation sequence from word beginWord to word endWord using a dictionary wordList is a sequence of words beginWord -> s1 -> s2 -> ... -> sk such that:

Every adjacent pair of words differs by a single letter.
Every si for 1 <= i <= k is in wordList. Note that beginWord does not need to be in wordList.
sk == endWord
Given two words, beginWord and endWord, and a dictionary wordList, return all the shortest transformation sequences from beginWord to endWord, or an empty list if no such sequence exists. Each sequence should be returned as a list of the words [beginWord, s1, s2, ..., sk].

 

Example 1:

Input: beginWord = "hit", endWord = "cog", wordList = ["hot","dot","dog","lot","log","cog"]
Output: [["hit","hot","dot","dog","cog"],["hit","hot","lot","log","cog"]]
Explanation: There are 2 shortest transformation sequences:
"hit" -> "hot" -> "dot" -> "dog" -> "cog"
"hit" -> "hot" -> "lot" -> "log" -> "cog"
Example 2:

Input: beginWord = "hit", endWord = "cog", wordList = ["hot","dot","dog","lot","log"]
Output: []
Explanation: The endWord "cog" is not in wordList, therefore there is no valid transformation sequence.
 

Constraints:

1 <= beginWord.length <= 5
endWord.length == beginWord.length
1 <= wordList.length <= 500
wordList[i].length == beginWord.length
beginWord, endWord, and wordList[i] consist of lowercase English letters.
beginWord != endWord
All the words in wordList are unique.
The sum of all shortest transformation sequences does not exceed 105.

### Solution

```cpp
class Solution {
public:
    vector<vector<string>> findLadders(string beginWord, string endWord, vector<string>& wordList) {
        vector<vector<string>> result;
        const int L = beginWord.size();
        const uint32_t MASK = 0x1F;
        
        auto encode = [&](const string& s) -> uint32_t {
            uint32_t code = 0;
            for (int i = 0; i < L; ++i) {
                code |= ((s[i] - 'a') + 1) << (5 * i);
            }
            return code;
        };
        
        unordered_map<uint32_t, int> codeToIdx;
        codeToIdx.reserve(wordList.size() * 2 + 1);
        
        vector<string> idxToWord;
        idxToWord.reserve(wordList.size() + 1);
        vector<uint32_t> codes;
        codes.reserve(wordList.size() + 1);
        
        int beginId, endId = -1;
        
        uint32_t beginCode = encode(beginWord);
        beginId = 0;
        codeToIdx[beginCode] = 0;
        idxToWord.push_back(beginWord);
        codes.push_back(beginCode);
        
        for (const string& w : wordList) {
            uint32_t c = encode(w);
            if (codeToIdx.find(c) != codeToIdx.end()) continue;
            
            if (w == endWord) endId = codes.size();
            codeToIdx[c] = codes.size();
            idxToWord.push_back(w);
            codes.push_back(c);
        }
        
        if (__builtin_expect(endId == -1, 0)) return result;
        
        const int N = codes.size();
        
        vector<int> dist(N, -1);
        vector<vector<short>> pred(N);
        
        for (auto& p : pred) p.reserve(4);
        
        queue<short> q;
        dist[beginId] = 0;
        q.push(beginId);
        
        while (__builtin_expect(!q.empty(), 1)) {
            short u = q.front(); q.pop();
            if (__builtin_expect(u == endId, 0)) break;
            
            uint32_t uCode = codes[u];
            int uDist = dist[u];
            
            for (int pos = 0; pos < L; ++pos) {
                uint32_t clearMask = ~(MASK << (5 * pos));
                uint32_t base = uCode & clearMask;
                
                for (int c = 1; c <= 26; ++c) {
                    uint32_t vCode = base | (c << (5 * pos));
                    if (__builtin_expect(vCode == uCode, 0)) continue;
                    
                    auto it = codeToIdx.find(vCode);
                    if (__builtin_expect(it == codeToIdx.end(), 0)) continue;
                    
                    short v = it->second;
                    
                    if (__builtin_expect(dist[v] == -1, 1)) {
                        dist[v] = uDist + 1;
                        pred[v].push_back(u);
                        q.push(v);
                    } else if (dist[v] == uDist + 1) {
                        pred[v].push_back(u);
                    }
                }
            }
        }
        
        if (dist[endId] == -1) return result;
        
        vector<short> path;
        path.reserve(dist[endId] + 1);
        
        function<void(short)> dfs = [&](short u) {
            path.push_back(u);
            if (__builtin_expect(u == beginId, 0)) {
                vector<string> res;
                res.reserve(path.size());
                for (auto it = path.rbegin(); it != path.rend(); ++it) {
                    res.push_back(idxToWord[*it]);
                }
                result.push_back(move(res));
            } else {
                for (short p : pred[u]) {
                    dfs(p);
                }
            }
            path.pop_back();
        };
        
        dfs(endId);
        return result;
    }
};
```
